-- ============================================================
-- 1. USERS TABLE (No dependencies)
-- ============================================================
CREATE TABLE Users (
    UserId          INT IDENTITY(1,1) PRIMARY KEY,
    Username        NVARCHAR(100) NOT NULL UNIQUE,
    Email           NVARCHAR(150) NOT NULL UNIQUE,
    PasswordHash    NVARCHAR(255) NOT NULL,
    FirstName       NVARCHAR(100) NULL,
    LastName        NVARCHAR(100) NULL,
    ProfileImage    NVARCHAR(255) NULL,
    Role            NVARCHAR(20) NOT NULL CHECK (Role IN ('Admin','Kids','General')),
    ParentUserId    INT NULL,
    IsActive        BIT NOT NULL DEFAULT 1,
    LastLoginDate   DATETIME2 NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT GETDATE(),
    UpdatedAt       DATETIME2 NULL,
    FOREIGN KEY (ParentUserId) REFERENCES Users(UserId)
);
GO

-- ============================================================
-- 2. CATEGORIES TABLE (No dependencies)
-- ============================================================
CREATE TABLE Categories (
    CategoryId          INT IDENTITY(1,1) PRIMARY KEY,
    Name                NVARCHAR(100) NOT NULL,
    Description         NVARCHAR(500) NULL,
    AudienceType        NVARCHAR(10) NOT NULL CHECK (AudienceType IN ('Kids','General')),
    ParentCategoryId    INT NULL,
    IconUrl             NVARCHAR(255) NULL,
    ColorCode           NVARCHAR(7) NULL,
    IsActive            BIT NOT NULL DEFAULT 1,
    SortOrder           INT NULL,
    CreatedAt           DATETIME2 NOT NULL DEFAULT GETDATE(),
    UpdatedAt           DATETIME2 NULL,
    FOREIGN KEY (ParentCategoryId) REFERENCES Categories(CategoryId)
);
GO

-- ============================================================
-- 3. VIDEOS TABLE (Depends on Users and Categories)
-- ============================================================
CREATE TABLE Videos (
    VideoId             INT IDENTITY(1,1) PRIMARY KEY,
    YouTubeVideoId      NVARCHAR(50) NOT NULL UNIQUE,
    Title               NVARCHAR(255) NOT NULL,
    Description         NVARCHAR(MAX) NULL,
    ThumbnailUrl        NVARCHAR(255) NULL,
    ChannelName         NVARCHAR(150) NULL,
    ChannelUrl          NVARCHAR(255) NULL,
    DurationSeconds     INT NULL,
    ViewsCount          BIGINT NULL DEFAULT 0,
    LikesCount          BIGINT NULL DEFAULT 0,
    CategoryId          INT NOT NULL,
    AudienceType        NVARCHAR(10) NOT NULL CHECK (AudienceType IN ('Kids','General')),
    Language            NVARCHAR(50) NULL,
    Tags                NVARCHAR(500) NULL,
    IsFeatured          BIT NOT NULL DEFAULT 0,
    IsPopular           BIT NOT NULL DEFAULT 0,
    Status              NVARCHAR(20) NOT NULL DEFAULT 'Pending' CHECK (Status IN ('Pending','Published','Rejected','Deleted','Archived')),
    PublishedAt         DATETIME2 NULL,
    CreatedBy           INT NOT NULL,
    CreatedAt           DATETIME2 NOT NULL DEFAULT GETDATE(),
    UpdatedAt           DATETIME2 NULL,
    DeletedAt           DATETIME2 NULL,
    FOREIGN KEY (CategoryId) REFERENCES Categories(CategoryId),
    FOREIGN KEY (CreatedBy) REFERENCES Users(UserId)
);
GO

-- ============================================================
-- 4. WATCHHISTORY TABLE (Depends on Users and Videos)
-- ============================================================
CREATE TABLE WatchHistory (
    WatchId          INT IDENTITY(1,1) PRIMARY KEY,
    UserId           INT NOT NULL,
    VideoId          INT NOT NULL,
    ProgressPercent  DECIMAL(5,2) NOT NULL DEFAULT 0,
    IsCompleted      BIT NOT NULL DEFAULT 0,
    LastPosition     INT NULL,
    DeviceType       NVARCHAR(50) NULL,
    WatchedAt        DATETIME2 NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (UserId) REFERENCES Users(UserId),
    FOREIGN KEY (VideoId) REFERENCES Videos(VideoId),
    CONSTRAINT UQ_UserVideoWatch UNIQUE (UserId, VideoId)
);
GO

-- ============================================================
-- 5. BOOKMARKS TABLE (Depends on Users and Videos)
-- ============================================================
CREATE TABLE Bookmarks (
    BookmarkId    INT IDENTITY(1,1) PRIMARY KEY,
    UserId        INT NOT NULL,
    VideoId       INT NOT NULL,
    Notes         NVARCHAR(500) NULL,
    CreatedAt     DATETIME2 NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (UserId) REFERENCES Users(UserId),
    FOREIGN KEY (VideoId) REFERENCES Videos(VideoId),
    CONSTRAINT UQ_UserVideoBookmark UNIQUE (UserId, VideoId)
);
GO

-- ============================================================
-- 6. RECOMMENDATIONS TABLE (Depends on Users and Videos)
-- ============================================================
CREATE TABLE Recommendations (
    RecommendationId  INT IDENTITY(1,1) PRIMARY KEY,
    VideoId           INT NOT NULL,
    SuggestedForUser  INT NULL,
    Reason            NVARCHAR(255) NULL,
    ConfidenceScore   DECIMAL(5,2) NULL,
    Status            NVARCHAR(20) NOT NULL DEFAULT 'Pending' CHECK (Status IN ('Pending','Approved','Rejected','Viewed')),
    CreatedAt         DATETIME2 NOT NULL DEFAULT GETDATE(),
    ViewedAt          DATETIME2 NULL,
    FOREIGN KEY (VideoId) REFERENCES Videos(VideoId),
    FOREIGN KEY (SuggestedForUser) REFERENCES Users(UserId)
);
GO

-- ============================================================
-- 7. ACTIVITYLOG TABLE (Depends on Users and Videos)
-- ============================================================
CREATE TABLE ActivityLog (
    ActivityId    INT IDENTITY(1,1) PRIMARY KEY,
    UserId        INT NOT NULL,
    ActivityType  NVARCHAR(30) NOT NULL CHECK (
        ActivityType IN (
            'Login', 'Logout', 'Play', 'Progress', 'Bookmark', 'Unbookmark',
            'Search', 'AddVideo', 'UpdateVideo', 'DeleteVideo', 'ViewRecommendation',
            'CompleteVideo', 'DashboardView', 'Share', 'Rate', 'Comment',
            'CategoryView', 'PopularView', 'RecentView'
        )
    ),
    VideoId       INT NULL,
    CategoryId    INT NULL,
    Details       NVARCHAR(500) NULL,
    IPAddress     NVARCHAR(50) NULL,
    UserAgent     NVARCHAR(255) NULL,
    SessionId     NVARCHAR(100) NULL,
    OccurredAt    DATETIME2 NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (UserId) REFERENCES Users(UserId),
    FOREIGN KEY (VideoId) REFERENCES Videos(VideoId),
    FOREIGN KEY (CategoryId) REFERENCES Categories(CategoryId)
);
GO

-- ============================================================
-- 8. VIDEORATINGS TABLE (Depends on Users and Videos)
-- ============================================================
CREATE TABLE VideoRatings (
    RatingId      INT IDENTITY(1,1) PRIMARY KEY,
    UserId        INT NOT NULL,
    VideoId       INT NOT NULL,
    Rating        TINYINT NOT NULL CHECK (Rating BETWEEN 1 AND 5),
    Review        NVARCHAR(500) NULL,
    CreatedAt     DATETIME2 NOT NULL DEFAULT GETDATE(),
    UpdatedAt     DATETIME2 NULL,
    FOREIGN KEY (UserId) REFERENCES Users(UserId),
    FOREIGN KEY (VideoId) REFERENCES Videos(VideoId),
    CONSTRAINT UQ_UserVideoRating UNIQUE (UserId, VideoId)
);
GO

-- Get Continue Watching
CREATE PROCEDURE sp_GetContinueWatching
    @UserId INT,
    @Limit INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    SELECT TOP (@Limit)
        v.VideoId, v.YouTubeVideoId, v.Title, v.ThumbnailUrl, v.ChannelName,
        v.DurationSeconds, v.ViewsCount, wh.ProgressPercent, wh.IsCompleted, wh.WatchedAt
    FROM WatchHistory wh
    INNER JOIN Videos v ON wh.VideoId = v.VideoId
    WHERE wh.UserId = @UserId AND wh.ProgressPercent < 100 AND wh.ProgressPercent > 0
        AND v.Status = 'Published'
    ORDER BY wh.WatchedAt DESC;
END
GO

-- Get Bookmarks
CREATE PROCEDURE sp_GetBookmarks
    @UserId INT,
    @Limit INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    SELECT TOP (@Limit)
        b.BookmarkId, v.VideoId, v.YouTubeVideoId, v.Title, v.ThumbnailUrl,
        v.ChannelName, v.DurationSeconds, v.ViewsCount, b.Notes, b.CreatedAt
    FROM Bookmarks b
    INNER JOIN Videos v ON b.VideoId = v.VideoId
    WHERE b.UserId = @UserId AND v.Status = 'Published'
    ORDER BY b.CreatedAt DESC;
END
GO

-- Get Recommendations
CREATE PROCEDURE sp_GetRecommendations
    @UserId INT,
    @Limit INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    SELECT TOP (@Limit)
        r.RecommendationId, v.VideoId, v.YouTubeVideoId, v.Title, v.ThumbnailUrl,
        v.ChannelName, v.DurationSeconds, v.ViewsCount, r.Reason, r.ConfidenceScore
    FROM Recommendations r
    INNER JOIN Videos v ON r.VideoId = v.VideoId
    WHERE r.SuggestedForUser = @UserId AND r.Status = 'Pending' AND v.Status = 'Published'
    ORDER BY r.ConfidenceScore DESC, r.CreatedAt DESC;
END
GO

-- Get Recent Videos
CREATE PROCEDURE sp_GetRecentVideos
    @AudienceType NVARCHAR(10) = NULL,
    @UserId INT = NULL,
    @Limit INT = 20,
    @Offset INT = 0
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        v.VideoId, v.YouTubeVideoId, v.Title, v.Description, v.ThumbnailUrl,
        v.ChannelName, v.DurationSeconds, v.ViewsCount, v.LikesCount, v.PublishedAt,
        c.Name as CategoryName,
        CASE WHEN b.BookmarkId IS NOT NULL THEN 1 ELSE 0 END as IsBookmarked,
        CASE WHEN wh.WatchId IS NOT NULL THEN 1 ELSE 0 END as IsWatched,
        wh.ProgressPercent as WatchProgress
    FROM Videos v
    INNER JOIN Categories c ON v.CategoryId = c.CategoryId
    LEFT JOIN Bookmarks b ON v.VideoId = b.VideoId AND b.UserId = @UserId
    LEFT JOIN WatchHistory wh ON v.VideoId = wh.VideoId AND wh.UserId = @UserId
    WHERE v.Status = 'Published' AND (@AudienceType IS NULL OR v.AudienceType = @AudienceType)
    ORDER BY v.PublishedAt DESC
    OFFSET @Offset ROWS FETCH NEXT @Limit ROWS ONLY;
END
GO

-- Get Categories

BEGIN

        c.CategoryId, c.Name, c.Description, c.IconUrl, c.ColorCode, c.AudienceType,

-- Search Videos
CREATE PROCEDURE sp_SearchVideos
    @SearchTerm NVARCHAR(255),
    @UserId INT = NULL,
    @AudienceType NVARCHAR(10) = NULL,
    @CategoryId INT = NULL,
    @Limit INT = 20,
    @Offset INT = 0
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        v.VideoId, v.YouTubeVideoId, v.Title, v.Description, v.ThumbnailUrl,
        v.ChannelName, v.DurationSeconds, v.ViewsCount, v.LikesCount, v.PublishedAt,
        c.Name as CategoryName,
        CASE WHEN b.BookmarkId IS NOT NULL THEN 1 ELSE 0 END as IsBookmarked,
        CASE WHEN wh.WatchId IS NOT NULL THEN 1 ELSE 0 END as IsWatched,
        wh.ProgressPercent as WatchProgress
    FROM Videos v
    INNER JOIN Categories c ON v.CategoryId = c.CategoryId
    LEFT JOIN Bookmarks b ON v.VideoId = b.VideoId AND b.UserId = @UserId
    LEFT JOIN WatchHistory wh ON v.VideoId = wh.VideoId AND wh.UserId = @UserId
    WHERE v.Status = 'Published'
        AND (v.Title LIKE '%' + @SearchTerm + '%' OR v.Description LIKE '%' + @SearchTerm + '%')
        AND (@AudienceType IS NULL OR v.AudienceType = @AudienceType)
        AND (@CategoryId IS NULL OR v.CategoryId = @CategoryId)
    ORDER BY v.ViewsCount DESC
    OFFSET @Offset ROWS FETCH NEXT @Limit ROWS ONLY;
END
GO

-- Play Video
CREATE PROCEDURE sp_PlayVideo
    @UserId INT,
    @VideoId INT,
    @DeviceType NVARCHAR(50) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM WatchHistory WHERE UserId = @UserId AND VideoId = @VideoId)
        UPDATE WatchHistory SET WatchedAt = GETDATE() WHERE UserId = @UserId AND VideoId = @VideoId;
    ELSE
        INSERT INTO WatchHistory (UserId, VideoId) VALUES (@UserId, @VideoId);
    UPDATE Videos SET ViewsCount = ViewsCount + 1 WHERE VideoId = @VideoId;
END
GO

-- Update Progress
CREATE PROCEDURE sp_UpdateProgress
    @UserId INT,
    @VideoId INT,
    @ProgressPercent DECIMAL(5,2)
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @IsCompleted BIT = CASE WHEN @ProgressPercent >= 100 THEN 1 ELSE 0 END;
    IF EXISTS (SELECT 1 FROM WatchHistory WHERE UserId = @UserId AND VideoId = @VideoId)
        UPDATE WatchHistory SET ProgressPercent = @ProgressPercent, IsCompleted = @IsCompleted, WatchedAt = GETDATE()
        WHERE UserId = @UserId AND VideoId = @VideoId;
    ELSE
        INSERT INTO WatchHistory (UserId, VideoId, ProgressPercent, IsCompleted) VALUES (@UserId, @VideoId, @ProgressPercent, @IsCompleted);
END
GO

-- Add Bookmark
CREATE PROCEDURE sp_AddBookmark
    @UserId INT,
    @VideoId INT,
    @Notes NVARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    IF NOT EXISTS (SELECT 1 FROM Bookmarks WHERE UserId = @UserId AND VideoId = @VideoId)
        INSERT INTO Bookmarks (UserId, VideoId, Notes) VALUES (@UserId, @VideoId, @Notes);
END
GO

-- Remove Bookmark
CREATE PROCEDURE sp_RemoveBookmark
    @UserId INT,
    @VideoId INT
AS
BEGIN
    SET NOCOUNT ON;
    DELETE FROM Bookmarks WHERE UserId = @UserId AND VideoId = @VideoId;
END
GO

-- Get Dashboard Stats
CREATE PROCEDURE sp_GetDashboardStats
    @UserId INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        (SELECT COUNT(*) FROM WatchHistory WHERE UserId = @UserId) as TotalWatched,
        (SELECT COUNT(*) FROM WatchHistory WHERE UserId = @UserId AND IsCompleted = 1) as CompletedVideos,
        (SELECT COUNT(*) FROM Bookmarks WHERE UserId = @UserId) as TotalBookmarks,
        (SELECT COUNT(*) FROM Recommendations WHERE SuggestedForUser = @UserId AND Status = 'Pending') as PendingRecommendations;
END
GO
